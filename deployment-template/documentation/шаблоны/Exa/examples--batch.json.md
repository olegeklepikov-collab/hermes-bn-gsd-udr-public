```json
{
  "requests": [
    {
      "customId": "official-search",
      "method": "POST",
      "url": "/search",
      "body": {
        "query": "${QUERY}",
        "type": "fast",
        "numResults": 1,
        "includeDomains": [
          "exa.ai/docs"
        ],
        "contents": {
          "highlights": true
        }
      }
    },
    {
      "customId": "paper-title",
      "method": "POST",
      "url": "/agent/runs",
      "body": {
        "query": "${QUERY}",
        "effort": "minimal"
      }
    }
  ],
  "metadata": {
    "purpose": "Hermes BN integration test"
  }
}
```
